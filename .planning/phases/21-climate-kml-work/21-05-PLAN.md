    ### Automated verification checks to run:

    **1. KML structure check:**
    ```bash
    python3 -c "
    import lxml.etree as ET
    tree = ET.parse('2050-snapshot/kml/climate.kml')
    ns = {'kml': 'http://www.opengis.net/kml/2.2'}
    folders = tree.findall('.//kml:Folder/kml:name', ns)
    fnames = [f.text for f in folders if f.text]
    print(f'Folders ({len(fnames)}): {fnames}')
    assert 'Köppen-Geiger Climate Classification (2050)' in fnames, 'Missing Köppen folder'
    assert 'Ecological Biomes (2050)' in fnames, 'Missing Biomes folder'
    assert 'Sea Level Rise Inundation (0.35m)' in fnames, 'Missing SLR folder'
    assert 'Climate' in fnames, 'Missing Climate folder'
    # Count total placemarks
    pms = tree.findall('.//kml:Placemark', ns)
    print(f'Total placemarks: {len(pms)}')
    # Climate folder placemarks
    for f in tree.findall('.//kml:Folder', ns):
        n = f.find('kml:name', ns)
        if n is not None and n.text == 'Climate':
            c_pms = f.findall('.//kml:Placemark', ns)
            print(f'Climate folder placemarks: {len(c_pms)}')
            pm_names = [pm.find('kml:name', ns).text for pm in c_pms if pm.find('kml:name', ns) is not None]
            for pn in pm_names:
                print(f'  - {pn}')
            assert len(c_pms) == 11, 'Expected 11 climate placemarks'
    # Check cross-references
    descs = tree.findall('.//kml:Placemark/kml:description', ns)
    see_refs = [d.text for d in descs if d.text and 'See:' in d.text]
    print(f'Placemarks with See: refs: {len(see_refs)}')
    assert len(see_refs) >= 11, f'Only {len(see_refs)} placemarks have cross-refs'
    print('ALL KML STRUCTURE CHECKS PASSED')
    "
    ```
