    python3 -c "
    import os, re, glob
    from lxml import etree
    
    ns = '{http://www.opengis.net/kml/2.2}'

    kml_names = {}
    for f in sorted(glob.glob('2050-snapshot/kml/[a-z]*.kml')):

    found_markers = 0

            # Check if any KML has a placemark with this name
            matched = any(m == name or m.startswith(name) or name.startswith(m) for name in all_kml_names)
            if matched:

            else:
                missing_markers.append(f'{doc}: {m}')
